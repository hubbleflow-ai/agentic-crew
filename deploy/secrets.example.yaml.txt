# Template only · the real Secret is created from .env by scripts/k8s-secrets.sh
# and is never committed.
#
# The .txt extension is deliberate. As a .yaml this file was picked up by
# `kubectl apply -f deploy/` and quietly overwrote the real Secret with
# "replace-me", which fails later and somewhere else -- agents booting fine and
# then getting 401 from Gemini. A template that can be applied by accident is a
# trap, so this one cannot be.
apiVersion: v1
kind: Secret
metadata:
  name: crew-secrets
  namespace: crew
type: Opaque
stringData:
  GOOGLE_API_KEY: "replace-me"
  E2B_API_KEY: ""
  LANGSMITH_API_KEY: ""
